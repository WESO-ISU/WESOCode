// test_load.ino
// Tests variable load functionality using the same pin definitions,
// loadLoadArray() and loadLoad() logic as the main firmware.
// Use a power supply and multimeter to verify correct resistance switching.
//
// HOW TO USE:
//   Open Serial Monitor at 9600 baud.
//   Enter a decimal value 0-255 and press send.
//   The load will switch to that byte pattern and report the expected resistance.
//   Use a multimeter across the load terminals to verify.

#include <stdint.h>

const int LOAD_PIN1 = 2;
const int LOAD_PIN2 = 3;
const int LOAD_PIN3 = 4;
const int LOAD_PIN4 = 5;
const int LOAD_PIN5 = 6;
const int LOAD_PIN6 = 7;
const int LOAD_PIN7 = 8;
const int LOAD_PIN8 = 9;

const bool LOAD_ACTIVE_HIGH = true; //#FLAG confirm with EE team

const int LOAD_PINS[8] = {
    LOAD_PIN8, LOAD_PIN7, LOAD_PIN6, LOAD_PIN5,
    LOAD_PIN4, LOAD_PIN3, LOAD_PIN2, LOAD_PIN1
};

// Resistor values matching the lookup table, MSB to LSB
const float RESISTOR_VALUES[8] = { 1280.0f, 640.0f, 320.0f, 168.0f, 82.0f, 41.0f, 23.5f, 11.0f };

int loadArray[8] = {0, 0, 0, 0, 0, 0, 0, 0};

// Exact copy from main firmware
void loadLoadArray(uint8_t loadValue) {
    for (int i = 0; i < 8; i++) {
        loadArray[i] = (loadValue >> (7 - i)) & 1;
    }
}

// Exact copy from main firmware
void loadLoad() {
    for (int i = 0; i < 8; i++) {
        bool activate = (loadArray[i] == 1);
        digitalWrite(LOAD_PINS[i], (activate == LOAD_ACTIVE_HIGH) ? HIGH : LOW);
    }
}

// Computes the expected parallel resistance for the current loadArray.
// Use this to verify what the multimeter should read.
float compute_parallel_resistance() {
    float conductance = 0.0f;
    bool any_on = false;
    for (int i = 0; i < 8; i++) {
        if (loadArray[i] == 1) {
            conductance += 1.0f / RESISTOR_VALUES[i];
            any_on = true;
        }
    }
    if (!any_on) return -1.0f; // open circuit
    return 1.0f / conductance;
}

void print_load_state(uint8_t loadValue) {
    Serial.print("Load byte: 0b");
    for (int i = 7; i >= 0; i--) Serial.print((loadValue >> i) & 1);
    Serial.print(" (");
    Serial.print(loadValue);
    Serial.println(")");

    Serial.print("Active resistors: ");
    bool any = false;
    for (int i = 0; i < 8; i++) {
        if (loadArray[i] == 1) {
            if (any) Serial.print(", ");
            Serial.print(RESISTOR_VALUES[i]);
            Serial.print("Ω");
            any = true;
        }
    }
    if (!any) Serial.print("none (open circuit)");
    Serial.println();

    float r = compute_parallel_resistance();
    if (r < 0) {
        Serial.println("Expected resistance: open circuit (∞ Ω)");
    } else {
        Serial.print("Expected parallel resistance: ");
        Serial.print(r, 4);
        Serial.println(" Ω  <-- verify with multimeter");
    }
    Serial.println("---");
}

void setup() {
    Serial.begin(9600);
    for (int i = 0; i < 8; i++) {
        pinMode(LOAD_PINS[i], OUTPUT);
        digitalWrite(LOAD_PINS[i], LOAD_ACTIVE_HIGH ? LOW : HIGH); // all off
    }
    Serial.println("Load test ready. Enter a value 0-255 in Serial Monitor.");
    Serial.println("Special values: 0=all off, 255=all on");
    Serial.println("---");
}

void loop() {
    if (Serial.available() > 0) {
        int input = Serial.parseInt();
        if (input < 0)   input = 0;
        if (input > 255) input = 255;

        // Flush remaining bytes (newline, etc.) so they don't
        // get parsed as a second value of 0
        while (Serial.available() > 0) Serial.read();

        uint8_t loadValue = (uint8_t)input;
        loadLoadArray(loadValue);
        loadLoad();
        print_load_state(loadValue);
    }
}
