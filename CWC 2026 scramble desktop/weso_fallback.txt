// WESO Fallback Firmware - Sweep & Hold Hill Climber
// -----------------------------------------------------------------------
// Requires no wind speed estimation, no lookup table, no sensor math.
// Sweeps a grid of LA duty + load byte combos, measures stabilized RPM
// at each, then locks in the best combo found and holds it indefinitely.
// Press RESWEEP_BTN_PIN (pin 26) at any time to trigger a fresh sweep.
// E-stop logic mirrors the main firmware exactly.
// -----------------------------------------------------------------------

#include <stdint.h>
#include <SPI.h>

// -----------------------------------------------------------------------
// PIN DEFINITIONS
// -----------------------------------------------------------------------
const int LOAD_PIN1 = 2;
const int LOAD_PIN2 = 3;
const int LOAD_PIN3 = 4;
const int LOAD_PIN4 = 5;
const int LOAD_PIN5 = 6;
const int LOAD_PIN6 = 7;
const int LOAD_PIN7 = 8;
const int LOAD_PIN8 = 9;

const int IR_PIN         = 15;
const int LA_PIN         = 0;
const int ESTOP_READ_PIN = 18;

const int LA_EXTEND_BTN_PIN  = 24;
const int LA_RETRACT_BTN_PIN = 25;
const int RESWEEP_BTN_PIN    = 26;   // Same pin as RECORD_BTN in main firmware
const int LA_GO_HOME_BTN_PIN = 27;

const int LOAD_PINS[8] = {
    LOAD_PIN8, LOAD_PIN7, LOAD_PIN6, LOAD_PIN5,
    LOAD_PIN4, LOAD_PIN3, LOAD_PIN2, LOAD_PIN1
};

// -----------------------------------------------------------------------
// LA CONSTANTS - constrained by optocouplers
// -----------------------------------------------------------------------
const int LA_DUTY_EXTEND  = 3805;   // Lower bound - most extended
const int LA_DUTY_RETRACT = 3880;   // Upper bound - most retracted
const int LA_DUTY_HOME    = 3825;   // Good starting position
const int E_STOP_DUTY     = 3880;   // Brake = full retract
const int LA_DUTY_MIN     = LA_DUTY_EXTEND;
const int LA_DUTY_MAX     = LA_DUTY_RETRACT;
const int LA_DUTY_STEP    = 1;

// -----------------------------------------------------------------------
// SWEEP PARAMETERS
// #FLAG - coarser steps = faster sweep but lower resolution.
//         Tune LA_SWEEP_STEP and LOAD_SWEEP_STEP after first test run.
// -----------------------------------------------------------------------
const int LA_SWEEP_STEP   = 5;    // Step size across LA_DUTY_MIN..LA_DUTY_MAX (~15 LA points)
const int LOAD_SWEEP_STEP = 16;   // Step size across 0..255 load byte (~16 load points)
                                  // Total combos: ~15 x 16 = ~240 points

// -----------------------------------------------------------------------
// SAFETY & TIMING CONSTANTS
// -----------------------------------------------------------------------
const float RPM_MAX_THRESHOLD       = 3000.0f;  //#FLAG verify safe max RPM with mechanical team
const float RPM_MIN_THRESHOLD       = 50.0f;    //#FLAG verify minimum meaningful spinning RPM
const float RPM_MARGIN              = 25.0f;    //#FLAG tune stability window after testing
const unsigned long RPM_STABLE_TIME_MS = 1500;  //#FLAG tune - ms RPM must hold before sampling
const unsigned long SETTLE_DELAY_MS    = 200;   // Extra settle time after each LA move

const bool LOAD_ACTIVE_HIGH = true; //#FLAG confirm with EE team before competition

// -----------------------------------------------------------------------
// STATE MACHINE
// -----------------------------------------------------------------------
enum State {
    STATE_ESTOP,
    STATE_STARTUP,
    STATE_SWEEP,
    STATE_HOLD
};

State current_state = STATE_STARTUP;

// -----------------------------------------------------------------------
// GLOBALS
// -----------------------------------------------------------------------
volatile unsigned long lastPulseTime  = 0;
volatile unsigned long pulseInterval  = 0;

int     la_current_duty   = LA_DUTY_HOME;
uint8_t current_load_byte = 0;
int     loadArray[8]      = {0, 0, 0, 0, 0, 0, 0, 0};

// Best combo found during sweep
int     best_la_duty   = LA_DUTY_HOME;
uint8_t best_load_byte = 0;
float   best_rpm       = 0.0f;

// Sweep state
int     sweep_la_duty   = LA_DUTY_MIN;
uint8_t sweep_load_byte = 0;

// -----------------------------------------------------------------------
// IR INTERRUPT
// FALLING edge - optically isolated signal is inverted.
// -----------------------------------------------------------------------
void ir_interrupt() {
    unsigned long now = micros();
    if (now - lastPulseTime < 10000) return;
    pulseInterval = now - lastPulseTime;
    lastPulseTime = now;
}

// -----------------------------------------------------------------------
// RPM
// -----------------------------------------------------------------------
float get_rpm() {
    noInterrupts();
    unsigned long interval  = pulseInterval;
    unsigned long lastPulse = lastPulseTime;
    interrupts();
    if (interval == 0) return 0.0f;
    if (micros() - lastPulse > 2000000) return 0.0f;
    return 60.0f / (interval / 1000000.0f);
}

// -----------------------------------------------------------------------
// LA CONTROL
// -----------------------------------------------------------------------
void la_write(int duty) {
    duty = constrain(duty, LA_DUTY_MIN, LA_DUTY_MAX);
    analogWriteResolution(12);
    analogWriteFrequency(LA_PIN, 50);
    analogWrite(LA_PIN, duty);
    la_current_duty = duty;
}

void la_step_toward(int target) {
    if (la_current_duty == target) return;
    int step = (target > la_current_duty) ? LA_DUTY_STEP : -LA_DUTY_STEP;
    la_write(la_current_duty + step);
}

void la_move_to_blocking(int target) {
    // Blocking ramp - used in e-stop only
    if (la_current_duty < target) {
        for (int i = la_current_duty; i <= target; i++) { la_write(i); delay(25); }
    } else {
        for (int i = la_current_duty; i >= target; i--) { la_write(i); delay(25); }
    }
}

// -----------------------------------------------------------------------
// LOAD CONTROL
// -----------------------------------------------------------------------
void loadLoadArray(uint8_t value) {
    for (int i = 0; i < 8; i++) {
        loadArray[i] = (value >> i) & 1;
    }
    current_load_byte = value;
}

void loadLoad() {
    for (int i = 0; i < 8; i++) {
        bool activate = (loadArray[i] == 1);
        digitalWrite(LOAD_PINS[i], (activate == LOAD_ACTIVE_HIGH) ? HIGH : LOW);
    }
}

void set_load(uint8_t value) {
    loadLoadArray(value);
    loadLoad();
}

// -----------------------------------------------------------------------
// SAFETY
// -----------------------------------------------------------------------
bool should_estop() {
    if (digitalRead(ESTOP_READ_PIN) == LOW) return true;
    if (get_rpm() > RPM_MAX_THRESHOLD)      return true;
    return false;
}

bool is_rpm_stable() {
    static float         last_rpm     = 0.0f;
    static unsigned long stable_since = 0;
    float rpm = get_rpm();
    if (abs(rpm - last_rpm) > RPM_MARGIN) {
        stable_since = millis();
    }
    last_rpm = rpm;
    return (millis() - stable_since >= RPM_STABLE_TIME_MS);
}

// -----------------------------------------------------------------------
// BUTTON HANDLING
// -----------------------------------------------------------------------
void handle_estop_button() {
    if (digitalRead(ESTOP_READ_PIN) == LOW) {
        Serial.println("E-stop button pressed!");
        current_state = STATE_ESTOP;
    }
}

void handle_manual_la_buttons() {
    if (current_state == STATE_SWEEP) return; // Don't allow manual LA during sweep
    bool extendPressed  = digitalRead(LA_EXTEND_BTN_PIN)  == LOW;
    bool retractPressed = digitalRead(LA_RETRACT_BTN_PIN) == LOW;
    if (extendPressed && la_current_duty - LA_DUTY_STEP >= LA_DUTY_MIN) {
        la_write(la_current_duty - LA_DUTY_STEP);
        Serial.print("LA duty: "); Serial.println(la_current_duty);
        delay(100);
    } else if (retractPressed && la_current_duty + LA_DUTY_STEP <= LA_DUTY_MAX) {
        la_write(la_current_duty + LA_DUTY_STEP);
        Serial.print("LA duty: "); Serial.println(la_current_duty);
        delay(100);
    }
}

void handle_la_go_home() {
    if (digitalRead(LA_GO_HOME_BTN_PIN) == LOW) {
        la_write(LA_DUTY_HOME);
        delay(50);
        Serial.println("LA going home.");
    }
}

void handle_resweep_button() {
    static bool last_state = HIGH;
    bool btn = digitalRead(RESWEEP_BTN_PIN);
    if (last_state == HIGH && btn == LOW) {
        Serial.println("Resweep requested - resetting sweep state.");
        sweep_la_duty   = LA_DUTY_MIN;
        sweep_load_byte = 0;
        best_rpm        = 0.0f;
        best_la_duty    = LA_DUTY_HOME;
        best_load_byte  = 0;
        set_load(0);
        current_state = STATE_SWEEP;
    }
    last_state = btn;
}

// -----------------------------------------------------------------------
// STATE: ESTOP
// -----------------------------------------------------------------------
void state_estop() {
    la_move_to_blocking(E_STOP_DUTY);
    set_load(0);

    static unsigned long release_started_ms = 0;
    bool  btn_released = (digitalRead(ESTOP_READ_PIN) == HIGH);
    float rpm          = get_rpm();

    Serial.print("E-STOP - btn_released: "); Serial.print(btn_released);
    Serial.print(" rpm: "); Serial.println(rpm);

    if (!btn_released)              { release_started_ms = 0; return; }
    if (rpm >= RPM_MAX_THRESHOLD)   { release_started_ms = 0; return; }
    if (release_started_ms == 0)    { release_started_ms = millis(); return; }

    if (millis() - release_started_ms >= 250) {
        Serial.println("E-stop released - resuming from STARTUP.");
        release_started_ms = 0;
        sweep_la_duty   = LA_DUTY_MIN;
        sweep_load_byte = 0;
        best_rpm        = 0.0f;
        best_la_duty    = LA_DUTY_HOME;
        best_load_byte  = 0;
        current_state   = STATE_STARTUP;
    }
}

// -----------------------------------------------------------------------
// STATE: STARTUP
// Home the LA, zero the load, wait for turbine to spin up.
// -----------------------------------------------------------------------
void state_startup() {
    la_write(LA_DUTY_HOME);
    set_load(0);

    float rpm = get_rpm();
    Serial.print("STARTUP - RPM: "); Serial.println(rpm);

    if (rpm > RPM_MIN_THRESHOLD) {
        Serial.println("Turbine spinning - starting sweep.");
        sweep_la_duty   = LA_DUTY_MIN;
        sweep_load_byte = 0;
        best_rpm        = 0.0f;
        current_state   = STATE_SWEEP;
    }
}

// -----------------------------------------------------------------------
// STATE: SWEEP
// Outer loop: LA duty (LA_DUTY_MIN -> LA_DUTY_MAX, step LA_SWEEP_STEP)
// Inner loop: load byte (0 -> 255, step LOAD_SWEEP_STEP)
// At each combo: wait for RPM stable, sample, record if best.
// Load is zeroed between LA steps to avoid stress at unknown combos.
// -----------------------------------------------------------------------
void state_sweep() {
    if (should_estop()) { current_state = STATE_ESTOP; return; }

    // Non-blocking: step LA toward target one step per loop iteration
    if (la_current_duty != sweep_la_duty) {
        la_step_toward(sweep_la_duty);
        delay(SETTLE_DELAY_MS);
        return;
    }

    // Apply current load
    set_load(sweep_load_byte);

    // Wait for RPM to stabilize before sampling
    if (!is_rpm_stable()) return;

    float rpm = get_rpm();
    Serial.print("SWEEP - LA: "); Serial.print(sweep_la_duty);
    Serial.print(" Load: ");      Serial.print(sweep_load_byte);
    Serial.print(" RPM: ");       Serial.println(rpm);

    if (rpm > best_rpm) {
        best_rpm        = rpm;
        best_la_duty    = sweep_la_duty;
        best_load_byte  = sweep_load_byte;
        Serial.print("  ^ New best! RPM="); Serial.print(best_rpm);
        Serial.print(" LA=");               Serial.print(best_la_duty);
        Serial.print(" Load=");             Serial.println(best_load_byte);
    }

    // Advance inner loop (load byte)
    if ((int)sweep_load_byte + LOAD_SWEEP_STEP <= 255) {
        sweep_load_byte += LOAD_SWEEP_STEP;
        return;
    }

    // Inner loop exhausted - zero load and advance outer loop (LA duty)
    set_load(0);
    sweep_load_byte = 0;

    if (sweep_la_duty + LA_SWEEP_STEP <= LA_DUTY_MAX) {
        sweep_la_duty += LA_SWEEP_STEP;
        return;
    }

    // All combos tested - move to HOLD
    Serial.println("===== SWEEP COMPLETE =====");
    Serial.print("Best LA duty:   "); Serial.println(best_la_duty);
    Serial.print("Best load byte: "); Serial.println(best_load_byte);
    Serial.print("Best RPM:       "); Serial.println(best_rpm);
    Serial.println("==========================");
    current_state = STATE_HOLD;
}

// -----------------------------------------------------------------------
// STATE: HOLD
// Locks in the best combo. Prints status every 2s.
// Press RESWEEP_BTN to re-sweep if wind conditions change.
// -----------------------------------------------------------------------
void state_hold() {
    if (should_estop()) { current_state = STATE_ESTOP; return; }

    la_write(best_la_duty);
    set_load(best_load_byte);

    static unsigned long last_print = 0;
    if (millis() - last_print >= 2000) {
        float rpm = get_rpm();
        Serial.print("HOLD - LA: ");   Serial.print(best_la_duty);
        Serial.print(" Load: ");        Serial.print(best_load_byte);
        Serial.print(" RPM: ");         Serial.println(rpm);
        last_print = millis();
    }
}

// -----------------------------------------------------------------------
// SETUP
// -----------------------------------------------------------------------
void setup() {
    Serial.begin(9600);

    pinMode(IR_PIN, INPUT_PULLUP);
    attachInterrupt(digitalPinToInterrupt(IR_PIN), ir_interrupt, FALLING);

    pinMode(LA_PIN, OUTPUT);
    la_write(LA_DUTY_HOME);
    delay(15);

    pinMode(LA_EXTEND_BTN_PIN,  INPUT_PULLUP);
    pinMode(LA_RETRACT_BTN_PIN, INPUT_PULLUP);
    pinMode(LA_GO_HOME_BTN_PIN, INPUT_PULLUP);
    pinMode(RESWEEP_BTN_PIN,    INPUT_PULLUP);
    pinMode(ESTOP_READ_PIN,     INPUT_PULLUP);

    for (int i = 0; i < 8; i++) {
        pinMode(LOAD_PINS[i], OUTPUT);
        digitalWrite(LOAD_PINS[i], LOAD_ACTIVE_HIGH ? LOW : HIGH);
    }

    Serial.println("WESO Fallback Firmware Ready. Waiting for turbine spin-up...");
}

// -----------------------------------------------------------------------
// LOOP
// -----------------------------------------------------------------------
void loop() {
    handle_estop_button();
    handle_manual_la_buttons();
    handle_la_go_home();
    handle_resweep_button();

    switch (current_state) {
        case STATE_ESTOP:   state_estop();   break;
        case STATE_STARTUP: state_startup(); break;
        case STATE_SWEEP:   state_sweep();   break;
        case STATE_HOLD:    state_hold();    break;
    }
}
